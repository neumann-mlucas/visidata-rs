# numeric_binning: `set-option` configures bin edges for freq-col, which we
# treat as no-op. Promoting set-option from no-op to real would unlock this one
# test but risks regressing many others we can't easily audit. Revisit when the
# options chunk (Phase 1.5 #7) ships.
numeric_binning

# xlsx-color-cells: tests VisiData's `--xlsx_meta_columns=True` option which
# emits per-cell metadata columns (fontColor, fillColor, cellPyObj) alongside
# values. Not shipping that mode; our xlsx loader is value-only.
xlsx-color-cells

# join-extend-mult: two-stage `extend` join across 3 sheets with VisiData's
# per-source column prefixing (data2_C, data3_E). Multi-stage name-prefix
# handling is a 50-LOC rabbit hole for one test; defer.
join-extend-mult

# load-sqlite-view: VisiData opens sqlite as a meta-sheet-of-tables, then
# `open-row キemp_view` enters that table. The app has this (parts list); the
# sheet-level replay driver doesn't.
load-sqlite-view

# issue964-inner-join: VisiData reads CSV as all-string, we auto-infer, so
# ISTAT codes like "082053" get read as int "82053" (leading zero lost).
# Right fix is CSV string-only + explicit type-* casts, but that regresses
# many currently-passing tests. Revisit when the Tier 1 Options chunk lands.
issue964-inner-join

# freq-fmtstr: output differs in VisiData's percent/histogram columns.
freq-fmtstr

# sortorder: output is the columns-sheet CONTENTS (name/width/type/fmtstr/...),
# not the parent sheet. Our columns-sheet is a no-op mode. Modelling the
# meta-sheet rows as parent's columns is a dedicated chunk (S-13 "columns-sheet"
# in Phase 1.5).
sortorder
